"""이미 학습된 체크포인트들을 UBFC-PHYS 로 cross-dataset 평가 (재학습 없이 평가만).

UBFC-PHYS 는 D:\\UBFC-PHYS_extracted\\sN\\sN\\vid_sN_T{1,2,3}.avi 구조로 압축 해제됨
(56명 전체 추출은 시간이 걸려 백그라운드로 진행 중 — 이 스크립트는 그 시점까지
추출된 subject 만큼만 사용해서 평가하고, 나중에 더 추출되면 다시 돌리면 됨).

영상 실측 fps ≈ 35.14Hz (UBFC-rPPG/PURE 의 30Hz 와 다름) — 모델 자체는 학습 시
고정된 fps=30 가정을 그대로 쓰지만(재학습 안 하므로), HR을 실제 BPM 단위로 정확히
환산하려면 evaluate_per_subject 의 fs 인자에는 UBFC-PHYS 의 실제 fps 를 넘겨야 한다.

사용:
  python scripts/eval_ubfc_phys_cross.py
"""
import os
import sys
import io
import json

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
try:
    sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding='utf-8', errors='replace')
except Exception:
    pass

import numpy as np
import torch

from src.models.bipulseformer import ViT_BiPulseFormer
from src.models.physformer_baseline import ViT_ST_ST_Compact3_TDC_gra_sharp
from src.data.rppg_dataset import get_dataloader
from src.evaluation import evaluate_per_subject

UBFC_PHYS_PATH = 'D:\\UBFC-PHYS_extracted'
UBFC_PHYS_FPS = 35.14  # 실측 (vid_s1_T1.avi 기준), subject마다 근사값


def eval_bipulseformer(ckpt, n_win, topk, routing_mode, label):
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model = ViT_BiPulseFormer(
        patches=(4, 4, 4), dim=96, ff_dim=144, num_heads=4, num_layers=12,
        dropout_rate=0.1, theta=0.7, image_size=(160, 128, 128),
        n_win=n_win, topk=topk, routing_mode=routing_mode,
    ).to(device)
    model.load_state_dict(torch.load(ckpt, map_location=device, weights_only=True), strict=False)
    model.eval()
    return _run(model, label, gra_sharp=2.0)


def eval_physformer(ckpt, label):
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model = ViT_ST_ST_Compact3_TDC_gra_sharp(
        patches=(4, 4, 4), dim=96, ff_dim=144, num_heads=4, num_layers=12,
        dropout_rate=0.1, theta=0.7, image_size=(160, 128, 128),
    ).to(device)
    model.load_state_dict(torch.load(ckpt, map_location=device, weights_only=True), strict=False)
    model.eval()
    return _run(model, label, gra_sharp=2.0)


def _run(model, label, gra_sharp):
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    loader = get_dataloader('UBFC-PHYS', UBFC_PHYS_PATH, batch_size=4, clip_len=160,
                            shuffle=False, chunk_step=80, face_crop=True,
                            data_type='diff_normalized', dynamic_detection_freq=0,
                            fps=UBFC_PHYS_FPS, num_workers=2)
    n_clips = len(loader.dataset)
    if n_clips == 0:
        print(f'[!] {label}: UBFC-PHYS 클립 0개 — 압축 해제 진행 상황 확인 필요')
        return None
    print(f'[*] {label}: UBFC-PHYS {n_clips} clips 평가 중...')
    all_p, all_g = [], []
    with torch.no_grad():
        for inputs, labels in loader:
            inputs = inputs.to(device, non_blocking=True)
            out = model(inputs, gra_sharp=gra_sharp)
            rPPG = out[0] if isinstance(out, tuple) else out
            rPPG = (rPPG - rPPG.mean(-1, keepdim=True)) / (rPPG.std(-1, keepdim=True) + 1e-8)
            all_p.append(rPPG.cpu().numpy()); all_g.append(labels.numpy())
    preds = np.concatenate(all_p); gts = np.concatenate(all_g)
    m = evaluate_per_subject(preds, gts, loader.dataset.samples,
                             fs=UBFC_PHYS_FPS, diff_flag=True, low_pass=0.75, high_pass=2.5)
    print(f"  n_subjects(task)={m['n_subjects']}  MAE={m['MAE_bpm']:.3f}  RMSE={m['RMSE_bpm']:.3f}  "
          f"MAPE={m['MAPE_pct']:.3f}%  Pearson={m['Pearson']:.4f}")
    return m


def main():
    targets = [
        ('BiPulseFormer fftfix (PURE→UBFC-rPPG trained)',
         'results/cross_82_pure_to_ubfc_fftfix/checkpoints/PURE_to_UBFC-rPPG_epoch3.pt',
         'bipulseformer', (1, 4, 4), 4, 'fft_power'),
        ('BiPulseFormer fftfix (UBFC-rPPG→PURE trained)',
         'results/cross_82_ubfc_to_pure_fftfix/checkpoints/UBFC-rPPG_to_PURE_epoch6.pt',
         'bipulseformer', (1, 4, 4), 4, 'fft_power'),
        ('PhysFormer baseline no-clip (PURE→UBFC-rPPG trained)',
         'results/cross_physformer_noclip/checkpoints/PURE_to_UBFC-rPPG_epoch5.pt',
         'physformer', None, None, None),
    ]
    results = {}
    for label, ckpt, kind, n_win, topk, routing_mode in targets:
        if not os.path.exists(ckpt):
            print(f'[!] missing checkpoint: {ckpt}')
            continue
        if kind == 'bipulseformer':
            m = eval_bipulseformer(ckpt, n_win, topk, routing_mode, label)
        else:
            m = eval_physformer(ckpt, label)
        if m is not None:
            results[label] = m

    print('\n' + '=' * 80)
    print(' UBFC-PHYS cross-dataset eval summary (재학습 없이, 기존 체크포인트로만 평가)')
    print('=' * 80)
    for label, m in results.items():
        print(f"  {label}: MAE={m['MAE_bpm']:.3f}  Pearson={m['Pearson']:.4f}  (n={m['n_subjects']})")

    os.makedirs('results/cross_ubfc_phys_eval', exist_ok=True)
    with open('results/cross_ubfc_phys_eval/summary.json', 'w', encoding='utf-8') as f:
        json.dump(results, f, indent=2)


if __name__ == '__main__':
    main()
